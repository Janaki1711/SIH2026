# ─────────────────────────────────────────────────────────────────────────────
# CMakeLists.txt — iTantra M3 Phase 1
#
# Build system for the Protobuf Communication Simulator.
#
# WHAT THIS FILE DOES:
#   1. Finds the protobuf package (compiler + runtime libraries).
#   2. Runs protoc on packet_schema.proto to generate C++ files:
#          generated/packet_schema.pb.h
#          generated/packet_schema.pb.cc
#   3. Compiles the encoder, decoder, and main.cpp.
#   4. Links against protobuf::libprotobuf.
#
# PREREQUISITES (vcpkg):
#   vcpkg install protobuf
#   Then configure CMake with:
#     cmake .. -DCMAKE_TOOLCHAIN_FILE=<path-to-vcpkg>/scripts/buildsystems/vcpkg.cmake
#
# BUILD COMMANDS:
#   mkdir build
#   cd build
#   cmake .. -DCMAKE_BUILD_TYPE=Release \
#             -DCMAKE_TOOLCHAIN_FILE=<vcpkg-root>/scripts/buildsystems/vcpkg.cmake
#   cmake --build . --config Release
#   ./itantra_phase1          (Linux/macOS)
#   .\Release\itantra_phase1.exe   (Windows)
#
# ANDROID NOTE:
#   In Phase 5 (Android integration), this CMakeLists.txt will be moved to:
#       app/src/main/cpp/CMakeLists.txt
#   and adapted with:
#     - find_library(log-lib log)    ← Android NDK log library
#     - add_library(itantra SHARED ...) ← shared library for JNI
#     - protobuf-lite instead of full protobuf
# ─────────────────────────────────────────────────────────────────────────────

cmake_minimum_required(VERSION 3.22)

project(
    itantra_m3
    VERSION 0.1.0
    DESCRIPTION "iTantra M3 Phase 1 — Protobuf Communication Simulator"
    LANGUAGES CXX
)

# ── C++ Standard ─────────────────────────────────────────────────────────────
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ── Compiler Warnings ────────────────────────────────────────────────────────
if(MSVC)
    add_compile_options(/W4 /WX /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()

# ── Protobuf ─────────────────────────────────────────────────────────────────
# find_package locates the protobuf runtime + compiler.
# When using vcpkg, this is handled automatically via the toolchain file.
find_package(Protobuf REQUIRED)

message(STATUS "Protobuf version  : ${Protobuf_VERSION}")
message(STATUS "Protobuf includes : ${Protobuf_INCLUDE_DIRS}")
message(STATUS "protoc            : ${Protobuf_PROTOC_EXECUTABLE}")

# System libprotobuf (v36) is built against abseil-cpp, and generated code can
# reference absl symbols directly — ld then requires the absl DSOs explicitly
# ("DSO missing from command line"), which FindProtobuf's protobuf::libprotobuf
# INTERFACE does not carry. Pull the full list from protobuf.pc when PkgConfig
# is available (vcpkg/config-mode protobuf exports its own transitive deps, so
# the extra entry is harmless there).
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(ITANTRA_PROTOBUF QUIET protobuf IMPORTED_TARGET)
endif()

set(ITANTRA_PROTOBUF_LIBS protobuf::libprotobuf)
if(TARGET PkgConfig::ITANTRA_PROTOBUF)
    list(APPEND ITANTRA_PROTOBUF_LIBS PkgConfig::ITANTRA_PROTOBUF)
endif()

# ── Proto Compilation ─────────────────────────────────────────────────────────
# protobuf_generate_cpp() calls protoc at build time.
# It produces:
#   ${PROTO_SRCS} — packet_schema.pb.cc
#   ${PROTO_HDRS} — packet_schema.pb.h
# Both are placed in ${CMAKE_CURRENT_BINARY_DIR}/generated/
set(PROTO_FILE "${CMAKE_CURRENT_SOURCE_DIR}/proto/packet_schema.proto")

# Create the generated directory
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")

# Generate protobuf C++ sources
# The PROTOC_OUT_DIR argument is available in protobuf 3.21+
# For older versions, we use the two-argument form which puts files in CMAKE_CURRENT_BINARY_DIR
protobuf_generate_cpp(PROTO_SRCS PROTO_HDRS ${PROTO_FILE})

# ── Core Sources ─────────────────────────────────────────────────────────────
set(ITANTRA_CORE_SRCS
    src/Member3Integration.cpp
    src/SemanticCompressor.cpp
    src/SemanticCodebook.cpp
    src/TinyMLAgent.cpp
    src/GeoResolver.cpp
    src/TranslationBridge.cpp
    src/PacketFramer.cpp
    src/PacketEncoder.cpp
    src/PacketDecoder.cpp
    ${PROTO_SRCS}
)

# ── Main Executable ──────────────────────────────────────────────────────────
# Needs the full core (PacketFramer, SemanticCompressor, …): PacketEncoder /
# PacketDecoder reference them, so listing only main+Encoder+Decoder left the
# link with undefined references.
add_executable(itantra_phase1
    src/main.cpp
    ${ITANTRA_CORE_SRCS}
)

# Include paths:
#   - src/          : for PacketEncoder.hpp, PacketDecoder.hpp
#   - build dir     : for packet_schema.pb.h (generated by protoc)
#   - protobuf hdrs : for google/protobuf/*.h
target_include_directories(itantra_phase1 PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
    "${CMAKE_CURRENT_BINARY_DIR}"
    "${Protobuf_INCLUDE_DIRS}"
)

target_link_libraries(itantra_phase1 PRIVATE
    ${ITANTRA_PROTOBUF_LIBS}
)

# ── Member 3 C++ Unit Tests ──────────────────────────────────────────────────
add_executable(itantra_m3_tests
    src/tests_member3.cpp
    ${ITANTRA_CORE_SRCS}
)

target_include_directories(itantra_m3_tests PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
    "${CMAKE_CURRENT_BINARY_DIR}"
    "${Protobuf_INCLUDE_DIRS}"
)

target_link_libraries(itantra_m3_tests PRIVATE
    ${ITANTRA_PROTOBUF_LIBS}
)

# Keep assert() active in the test binary even in Release: -DNDEBUG would
# compile every verification away (the suite would print PASS while checking
# nothing) and turn flagged-but-only-inside-assert variables into -Werror.
target_compile_options(itantra_m3_tests PRIVATE
    $<$<CONFIG:Release,RelWithDebInfo>:-UNDEBUG>
)

# ── Android NDK / JNI Shared Library (libitantra_core.so) ────────────────────
if(ANDROID)
    find_library(log-lib log)
    add_library(itantra_core SHARED
        src/native_bridge.cpp
        ${ITANTRA_CORE_SRCS}
    )
    target_include_directories(itantra_core PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/src"
        "${CMAKE_CURRENT_BINARY_DIR}"
        "${Protobuf_INCLUDE_DIRS}"
    )
    target_link_libraries(itantra_core PRIVATE
        ${log-lib}
        protobuf::libprotobuf-lite
    )
else()
    find_package(JNI QUIET)
    if(JNI_FOUND)
        add_library(itantra_core SHARED
            src/native_bridge.cpp
            ${ITANTRA_CORE_SRCS}
        )
        target_include_directories(itantra_core PRIVATE
            "${CMAKE_CURRENT_SOURCE_DIR}/src"
            "${CMAKE_CURRENT_BINARY_DIR}"
            "${Protobuf_INCLUDE_DIRS}"
            ${JNI_INCLUDE_DIRS}
        )
        target_link_libraries(itantra_core PRIVATE
            ${ITANTRA_PROTOBUF_LIBS}
        )
    endif()
endif()

# ── Windows-specific: ensure console uses UTF-8 ──────────────────────────────
if(WIN32)
    # /utf-8 tells MSVC to interpret source files as UTF-8
    # The runtime chcp 65001 call in main.cpp handles console output
    target_compile_options(itantra_phase1 PRIVATE /utf-8)
    target_compile_options(itantra_m3_tests PRIVATE /utf-8)
endif()

# ── Status Output ─────────────────────────────────────────────────────────────
message(STATUS "")
message(STATUS "iTantra M3 build configuration:")
message(STATUS "  C++ standard  : ${CMAKE_CXX_STANDARD}")
message(STATUS "  Build type    : ${CMAKE_BUILD_TYPE}")
message(STATUS "  Proto file    : ${PROTO_FILE}")
message(STATUS "  Generated dir : ${CMAKE_CURRENT_BINARY_DIR}")
message(STATUS "")
